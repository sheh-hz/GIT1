#include <stdio.h>

void bubbleSort(int *p,int l)
{
    int i,j;
    for(i=0;i<l;i++)
    {
        for(j=0;j<l-1;j++)
        {
            if(*(p+j)>*(p+j+1))
            {
                int t=*(p+j);
                *(p+j)=*(p+j+1);
                *(p+j+1)=t;
            }
        }
    }
    for(i=0;i<l;i++)
    {
        printf("%d",*(p+i));
    }
}
void insertionSort(int *p,int l)
{
    int i, j, key;

    for(i = 1; i < l; i++)
    {
        key = *(p + i);
        j = i - 1;

        while(j >= 0 && *(p + j) > key)
        {
            *(p + j + 1) = *(p + j);
            j--;
        }

        *(p + j + 1) = key;
    }

    printf("Sorted array: ");
    for(i = 0; i < l; i++)
        printf("%d ", *(p + i));
}
void selectionSort(int *p,int l)
{
    int i, j, min, temp;

    for(i = 0; i < l - 1; i++)
    {
        min = i;

        for(j = i + 1; j < l; j++)
        {
            if(*(p + j) < *(p + min))
                min = j;
        }

        if(min != i)
        {
            temp = *(p + i);
            *(p + i) = *(p + min);
            *(p + min) = temp;
        }
    }

    printf("Sorted array: ");
    for(i = 0; i < l; i++)
        printf("%d\t", *(p + i));
}
int main() {
    int arr[10],choice,i;
    void (*operation)(int*,int);

    printf("Enter array: ");
    for(i=0;i<5;i++)
    {
        scanf("%d",&arr[i]);
    }

    printf("\nChoose an operation:\n");
    printf("1. bubbleSort\n");
    printf("2. insertionSort\n");
    printf("3. selectionSort\n");
    
    printf("Enter your choice: ");
    scanf("%d", &choice);

    switch (choice) {
        case 1:
            operation = bubbleSort;
            break;
        case 2:
            operation = insertionSort;
            break;
        case 3:
            operation = selectionSort;
            break;
        default:
            printf("Invalid choice!\n");
            return 1;
    }

     (*operation)(arr, 5);

    return 0;
}
