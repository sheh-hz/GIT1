#include <stdio.h>

int isEven(int n)
{
    return n % 2 == 0;
}

int isPositive(int n)
{
    return n > 0;
}

int isPrime(int n)
{
    int i;

    if(n <= 1)
        return 0;

    for(i = 2; i < n; i++)
    {
        if(n % i == 0)
            return 0;
    }

    return 1;
}

int find(int arr[], int size, int (*condition)(int))
{
    int i;

    for(i = 0; i < size; i++)
    {
        if(condition(arr[i]))
            return arr[i];
    }

    return -1;
}

int main()
{
    int arr[] = {-3, -1, 4, 7, 9, 11};
    int size = 6;
    int result;

    result = find(arr, size, isEven);
    printf("First even element: %d\n", result);

    result = find(arr, size, isPositive);
    printf("First positive element: %d\n", result);

    result = find(arr, size, isPrime);
    printf("First prime element: %d\n", result);

    return 0;
}
