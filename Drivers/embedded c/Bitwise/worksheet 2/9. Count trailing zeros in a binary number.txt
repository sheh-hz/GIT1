#include <stdio.h>
int trailing(int num)
{
    int Count=0,i;
    for(i=0;i<32;i++)
    {
        if((num&(1<<i))==0)
        {
          Count++;
        }
        else
        {
            break;
        }
    }
    return Count;
}


int main()
{
    int a=40;
    
    printf("trailing zeros: %d",trailing(a));
    return 0;
}
