#include <stdio.h>
#include <stdlib.h>

int main(int argc, char *argv[])
{
    FILE *fp;
    int sum = 0;
    int i;

    if (argc < 2)
    {
        printf("Please provide numbers as arguments\n");
        return 1;
    }

    for (i = 1; i < argc; i++)
    {
        sum += atoi(argv[i]);
    }

    fp = fopen("sum.txt", "w");

    if (fp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    fprintf(fp, "%d\n", sum);

    fclose(fp);

    printf("Sum = %d\n", sum);
    printf("Result written to sum.txt\n");

    return 0;
}
