#include <stdio.h>
#include <ctype.h>

int main()
{
    FILE *fp;
    char filename[50];
    char ch;

    printf("Enter file name: ");
    scanf("%s", filename);

    fp = fopen(filename, "r");

    if (fp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    printf("\nUppercase content:\n");

    while ((ch = fgetc(fp)) != EOF)
    {
        ch = toupper((unsigned char)ch);
        putchar(ch);
    }

    fclose(fp);

    return 0;
}
