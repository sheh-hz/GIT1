#include <stdio.h>
#include <string.h>

int main()
{
    FILE *fp;
    char filename[50];
    char word[50];
    char replacement[50];
    char line[500];

    printf("Enter file name: ");
    scanf("%s", filename);

    printf("Enter word to replace: ");
    scanf("%s", word);

    printf("Enter replacement: ");
    scanf("%s", replacement);

    fp = fopen(filename, "r");

    if (fp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    printf("\nContent after replacement:\n");

    while (fgets(line, sizeof(line), fp) != NULL)
    {
        char *pos;

        while ((pos = strstr(line, word)) != NULL)
        {
            char temp[500];

            strcpy(temp, pos + strlen(word));

            strcpy(pos, replacement);

            strcat(line, temp);
        }

        printf("%s", line);
    }

    fclose(fp);

    return 0;
}
