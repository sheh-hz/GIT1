#include <stdio.h>
#include <stdlib.h>
void main()
{
    int **p,i;
    *p=malloc(sizeof(int)*5);
    for(i=0;i<5;i++)
    {
        *(*p+i)=i;
    }
    for(i=0;i<5;i++)
    {
        printf("%d\t",*(*p+i));
    }
    *(*p+3)=100;
    printf("\n");
    for(i=0;i<5;i++)
    {
        printf("%d\t",*(*p+i));
    }
    
}
