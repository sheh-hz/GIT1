#include <stdio.h>
#include <string.h>

int main()
{
    FILE *fp;
    char line[200];
    int count = 0;
    int length;

    fp = fopen("data.txt", "r");

    if (fp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    while (fgets(line, sizeof(line), fp) != NULL)
    {
        length = strlen(line);

        /* Remove newline */
        if (length > 0 && line[length - 1] == '\n')
        {
            line[length - 1] = '\0';
            length--;
        }

        /* Check last character */
        if (length > 0 && line[length - 1] == ';')
        {
            count++;
        }
    }

    fclose(fp);

    printf("Lines ending with semicolon: %d\n", count);

    return 0;
}
