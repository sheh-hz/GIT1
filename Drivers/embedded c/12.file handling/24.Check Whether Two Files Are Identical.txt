#include <stdio.h>

int main()
{
    FILE *fp1, *fp2;
    char ch1, ch2;
    int different = 0;

    fp1 = fopen("a.txt", "r");
    fp2 = fopen("b.txt", "r");

    if (fp1 == NULL || fp2 == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    while (1)
    {
        ch1 = fgetc(fp1);
        ch2 = fgetc(fp2);

        if (ch1 != ch2)
        {
            different = 1;
            break;
        }

        if (ch1 == EOF && ch2 == EOF)
            break;
    }

    fclose(fp1);
    fclose(fp2);

    if (different)
        printf("Files are different\n");
    else
        printf("Files are identical\n");

    return 0;
}
