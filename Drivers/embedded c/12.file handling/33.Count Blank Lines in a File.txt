#include <stdio.h>
#include <string.h>

int main()
{
    FILE *fp;
    char line[200];
    int blankLines = 0;

    fp = fopen("data.txt", "r");

    if (fp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    while (fgets(line, sizeof(line), fp) != NULL)
    {
        if (strcmp(line, "\n") == 0 ||
            strcmp(line, "\r\n") == 0)
        {
            blankLines++;
        }
    }

    fclose(fp);

    printf("Blank lines: %d\n", blankLines);

    return 0;
}
