#include <stdio.h>

int main()
{
    FILE *fp, *temp;
    char filename[50];
    char line[200];
    int removeLine;
    int currentLine = 1;

    printf("Enter file name: ");
    scanf("%s", filename);

    printf("Enter line to remove: ");
    scanf("%d", &removeLine);

    fp = fopen(filename, "r");
    temp = fopen("temp.txt", "w");

    if (fp == NULL || temp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    while (fgets(line, sizeof(line), fp) != NULL)
    {
        if (currentLine != removeLine)
        {
            fputs(line, temp);
        }

        currentLine++;
    }

    fclose(fp);
    fclose(temp);

    remove(filename);
    rename("temp.txt", filename);

    printf("Line %d removed successfully.\n", removeLine);

    return 0;
}
