#include <stdio.h>
#include <string.h>

int main()
{
    FILE *fp, *temp;

    char filename[50];
    char line[200];
    char newText[200];

    int lineNumber;
    int currentLine = 1;

    printf("Enter file name: ");
    scanf("%s", filename);

    printf("Enter line number: ");
    scanf("%d", &lineNumber);

    getchar();

    printf("Enter new text: ");
    fgets(newText, sizeof(newText), stdin);

    fp = fopen(filename, "r");
    temp = fopen("temp.txt", "w");

    if (fp == NULL || temp == NULL)
    {
        printf("Unable to open file\n");
        return 1;
    }

    while (fgets(line, sizeof(line), fp) != NULL)
    {
        if (currentLine == lineNumber)
        {
            fputs(newText, temp);
        }
        else
        {
            fputs(line, temp);
        }

        currentLine++;
    }

    fclose(fp);
    fclose(temp);

    remove(filename);
    rename("temp.txt", filename);

    printf("Line %d changed successfully.\n", lineNumber);

    return 0;
}
