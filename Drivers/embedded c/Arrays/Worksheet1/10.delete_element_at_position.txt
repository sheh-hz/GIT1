#include <stdio.h>

int main()
{
    int num,i,pos;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num+1];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    printf("Delete at position: : ");
    scanf("%d",&pos);
    
    for(i=pos;i<num;i++)
    {
        arr[i]=arr[i+1];
        num--;
    }
    
    for(i=0;i<num;i++)
    {
        printf("%d\t",arr[i]);
    }
    
    return 0;
}
