#include <stdio.h>

int main()
{
    int num,i,sum=0;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<num;i++)
    {
        if(arr[i]%2 == 0)
        {
            sum+=arr[i];
        }
    }
    printf("sum of even numbers:%d",sum);
    return 0;
}
