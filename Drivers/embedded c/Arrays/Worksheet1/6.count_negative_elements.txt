#include <stdio.h>

int main()
{
    int num,i,c1=0;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<num;i++)
    {
        if(arr[i]<0)
        {
            c1++;
        }
    }
    printf("Total negative elements: %d",c1);
    return 0;
}
