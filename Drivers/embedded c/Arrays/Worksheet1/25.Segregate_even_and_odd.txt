#include <stdio.h>

int main()
{
    int num,i,count=0,j,e=0,o=0;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<num;i++)
    {
        if(arr[i]%2==0)
            e++;
        else
            o++;
    }
    int odd[num],even[num];
    int c1=0,c2=0;
    for(i=0;i<num;i++)
    {
        if(arr[i]%2==0){
            even[c1]=arr[i];c1++;}
        else{
            odd[c2]=arr[i];c2++;}
    }
    
    printf("even  : ");
    for(i=0;i<e;i++)
    {
        printf("%d\t",even[i]);
    }
    printf("\nodd  : ");
    for(i=0;i<o;i++)
    {
        printf("%d\t",odd[i]);
    }
    return 0;
}
