#include <stdio.h>

int main()
{
    int num,i,sum=0;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    int max=arr[0],min=arr[0];
    
    for(i=1;i<num;i++)
    {
        if(max<arr[i])
        {
            max=arr[i];
        }
        if(min>arr[i])
        {
            min=arr[i];
        }
    }
    printf("Maximum = %d\nMinimum = %d",max,min);
    return 0;
}
