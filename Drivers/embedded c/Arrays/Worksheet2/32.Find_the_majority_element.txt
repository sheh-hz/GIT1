#include <stdio.h>

int main()
{
    int n, i, j, count;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int arr[n];

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    for(i = 0; i < n; i++)
    {
        count = 0;

        for(j = 0; j < n; j++)
        {
            if(arr[i] == arr[j])
                count++;
        }

        if(count > n / 2)
        {
            printf("Majority element: %d", arr[i]);
            return 0;
        }
    }

    printf("No majority element");

    return 0;
}
