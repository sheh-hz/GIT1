#include <stdio.h>

int main()
{
    int n, i, j, sum, count = 0;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    int arr[n];

    printf("Enter array elements: ");
    for(i = 0; i < n; i++)
        scanf("%d", &arr[i]);

    printf("Enter sum: ");
    scanf("%d", &sum);

    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(arr[i] + arr[j] == sum)
                count++;
        }
    }

    printf("Number of pairs: %d", count);

    return 0;
}
