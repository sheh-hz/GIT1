#include <stdio.h>

int main()
{
    int arr[] = {10, 20, 10, 30, 20, 20, 10, 10};
    int n = 8;

    int visited[8] = {0};

    int i, j;
    int count;

    for (i = 0; i < n; i++)
    {
        if (visited[i] == 1)
            continue;

        count = 1;

        for (j = i + 1; j < n; j++)
        {
            if (arr[i] == arr[j])
            {
                count++;
                visited[j] = 1;
            }
        }

        printf("%d: %d\n", arr[i], count);
    }

    return 0;
}
