#include <stdio.h>

int main()
{
    int arr[] = {1, 2, 3, -4, -1, 4};
    int n = 6;

    int temp[6];
    int positive[6];
    int negative[6];

    int p = 0, neg = 0;
    int i, index = 0;

    for (i = 0; i < n; i++)
    {
        if (arr[i] < 0)
            negative[neg++] = arr[i];
        else
            positive[p++] = arr[i];
    }

    /* Start with negative number */
    i = 0;

    while (i < neg && i < p)
    {
        temp[index++] = negative[i];
        temp[index++] = positive[i];
        i++;
    }

    /* Remaining negative numbers */
    while (i < neg)
        temp[index++] = negative[i++];

    /* Remaining positive numbers */
    while (i < p)
        temp[index++] = positive[i++];

    printf("Rearranged array: ");

    for (i = 0; i < n; i++)
        printf("%d ", temp[i]);

    return 0;
}
