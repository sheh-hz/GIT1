#include <stdio.h>

int main()
{
    int arr[] = {100, 4, 200, 1, 3, 2};
    int n = 6;

    int i, j, temp;
    int count = 1;
    int maxCount = 1;

    /* Sort array */
    for (i = 0; i < n - 1; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            if (arr[i] > arr[j])
            {
                temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }

    /* Find consecutive sequence */
    for (i = 1; i < n; i++)
    {
        if (arr[i] == arr[i - 1] + 1)
        {
            count++;

            if (count > maxCount)
                maxCount = count;
        }
        else if (arr[i] != arr[i - 1])
        {
            count = 1;
        }
    }

    printf("Longest consecutive sequence length: %d\n",
           maxCount);

    return 0;
}
