#include <stdio.h>

int main()
{
    int arr[] = {0, 1, 2, 0, 1, 2, 1, 0};
    int n = 8;

    int count0 = 0;
    int count1 = 0;
    int count2 = 0;

    int i, index = 0;

    for (i = 0; i < n; i++)
    {
        if (arr[i] == 0)
            count0++;
        else if (arr[i] == 1)
            count1++;
        else
            count2++;
    }

    while (count0--)
        arr[index++] = 0;

    while (count1--)
        arr[index++] = 1;

    while (count2--)
        arr[index++] = 2;

    printf("Sorted array: ");

    for (i = 0; i < n; i++)
        printf("%d ", arr[i]);

    return 0;
}
