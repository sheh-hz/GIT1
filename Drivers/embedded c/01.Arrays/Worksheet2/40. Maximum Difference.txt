#include <stdio.h>

int main()
{
    int arr[] = {2, 3, 10, 6, 4, 8, 1};
    int n = 7;

    int i, j;
    int difference;
    int maxDifference = arr[1] - arr[0];

    for (i = 0; i < n - 1; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            difference = arr[j] - arr[i];

            if (difference > maxDifference)
                maxDifference = difference;
        }
    }

    printf("Maximum difference: %d\n", maxDifference);

    return 0;
}
