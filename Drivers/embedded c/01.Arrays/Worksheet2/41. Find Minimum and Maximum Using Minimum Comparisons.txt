#include <stdio.h>

int main()
{
    int arr[] = {1000, 11, 445, 1, 330, 3000};
    int n = 6;

    int min, max;
    int i;
    int a, b;

    if (n % 2 == 0)
    {
        if (arr[0] < arr[1])
        {
            min = arr[0];
            max = arr[1];
        }
        else
        {
            min = arr[1];
            max = arr[0];
        }

        i = 2;
    }
    else
    {
        min = max = arr[0];
        i = 1;
    }

    while (i < n - 1)
    {
        a = arr[i];
        b = arr[i + 1];

        if (a < b)
        {
            if (a < min)
                min = a;

            if (b > max)
                max = b;
        }
        else
        {
            if (b < min)
                min = b;

            if (a > max)
                max = a;
        }

        i = i + 2;
    }

    printf("Minimum element is %d\n", min);
    printf("Maximum element is %d\n", max);

    return 0;
}
