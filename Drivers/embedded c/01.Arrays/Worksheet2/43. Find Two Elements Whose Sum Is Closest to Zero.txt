#include <stdio.h>
#include <stdlib.h>

int main()
{
    int arr[] = {1, 60, -10, 70, -80, 85};
    int n = 6;

    int i, j;
    int sum;
    int minSum;
    int first, second;

    minSum = arr[0] + arr[1];
    first = arr[0];
    second = arr[1];

    for (i = 0; i < n - 1; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            sum = arr[i] + arr[j];

            if (abs(sum) < abs(minSum))
            {
                minSum = sum;
                first = arr[i];
                second = arr[j];
            }
        }
    }

    printf("Pair whose sum is closest to zero: %d %d\n",
           first, second);

    return 0;
}
