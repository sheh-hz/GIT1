#include <stdio.h>

int main()
{
    int num,i,sum=0;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Enter the elements: ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<num;i++)
    {
        sum+=arr[i];
    }
    printf("sum = %d",sum);
    return 0;
}
