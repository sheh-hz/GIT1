#include <stdio.h>

int main()
{
    int num,i,count=0,j,k;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num],freq[num];
    
    for (int i = 0; i < num; i++)
        freq[i] = 0;
        
    printf("array : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    
    for(i=0;i<num;i++)
    {
        if(freq[i]==1)
            continue;
        count=1;
        for(j=i+1;j<num;)
        {
            if(arr[i]==arr[j])
            {
                
                for (int k = j; k < num - 1; k++) 
                {
                    arr[k] = arr[k + 1];
                }
                num--;
            } else {
                j++;
            }
        }
        printf("%d ",arr[i]);
    }
    
    return 0;
}
