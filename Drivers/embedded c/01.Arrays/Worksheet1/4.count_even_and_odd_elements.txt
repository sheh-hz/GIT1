#include <stdio.h>

int main()
{
    int num,i,sum=0,c1=0,c2=0;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<num;i++)
    {
        if(arr[i]%2==0)
        {
            c1++;
        }
        else
        {
            c2++;
        }
    }
    printf("Even = %d\nOdd = %d",c1,c2);
    return 0;
}
