#include <stdio.h>

int main()
{
    int num,i,count=0,j;
    printf("Enter the number of elements:");
    scanf("%d",&num);
    int arr[num];
    printf("Input : ");
    for(i=0;i<num;i++)
    {
        scanf("%d",&arr[i]);
    }
    for(i=0;i<num;i++)
    {
        for(j=i+1;j<num;j++)
        {
            if(arr[i]==arr[j])
            {
                count++;
                break;
            }
        }
    }
    printf("count: %d",count);
    return 0;
}
