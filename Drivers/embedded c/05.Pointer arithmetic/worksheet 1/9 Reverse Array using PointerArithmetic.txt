#include <stdio.h>

int main() 
{
    
    int arr[] = {1, 2, 3, 4, 5},i;
    int len=sizeof(arr)/sizeof(arr[0]);
    int *p=arr;
    int *q=&arr[len-1];
    while(p<q)
    {
        int temp=*p;
        *p=*q;
        *q=temp;
        p++;
        q--;
    }
    for(i=0;i<len;i++)
    {
        printf("%d ",arr[i]);
    }
    return 0;
}
