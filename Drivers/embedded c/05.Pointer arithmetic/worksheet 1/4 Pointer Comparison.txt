#include <stdio.h>

int main() 
{
    
    int arr[] = {11, 22, 33, 44, 55};
    int *ptr1 = &arr[2];  // points to 33
    int *ptr2 = &arr[4];  // points to 55
    if(ptr1<ptr2)
    {
        printf("Pointer ptr1 points to an earlier element than ptr2");
    }
    else
        printf("Pointer ptr2 points to an earlier element than ptr1");
    return 0;
}
