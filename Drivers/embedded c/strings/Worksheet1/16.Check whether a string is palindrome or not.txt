#include <stdio.h>
#include <string.h>

int main()
{
    char a[20],b[20];
    int i, len,j=0;

    scanf("%[^\n]", a);
    len=strlen(a);
    
    for(i=len-1;i>=0;i--)
    {
        b[j]=a[i];
        j++;
        
    }
    b[j]='\0';
    if((strcmp(a,b))==0)
        printf("palindrome");
    else
        printf("not palindrome");

    return 0;
}
