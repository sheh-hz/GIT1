#include <stdio.h>
#include <string.h>

int main()
{
    char a[200], word[50];
    int i, len, wlen;

    scanf("%[^\n]", a);
    scanf(" %[^\n]", word);

    wlen = strlen(word);

    while (strstr(a, word) != NULL)
    {
        char *pos = strstr(a, word);
        len = strlen(a);

        for (i = pos - a; i <= len - wlen; i++)
        {
            a[i] = a[i + wlen];
        }
    }

    printf("%s", a);

    return 0;
}
