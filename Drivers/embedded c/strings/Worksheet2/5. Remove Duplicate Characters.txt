#include <stdio.h>

int main()
{
    char a[] = "programming";
    char b[100];
    int c[256] = {0};
    int i, j = 0;

    for (i = 0; a[i] != '\0'; i++)
    {
        if (c[(unsigned char)a[i]] == 0)
        {
            b[j++] = a[i];
            c[(unsigned char)a[i]] = 1;
        }
    }

    b[j] = '\0';

    printf("%s\n", b);

    return 0;
}
