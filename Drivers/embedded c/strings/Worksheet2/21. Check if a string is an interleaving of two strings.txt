#include <stdio.h>
#include <string.h>

int isInterleaving(char s1[], char s2[], char s3[])
{
    int n1 = strlen(s1);
    int n2 = strlen(s2);
    int n3 = strlen(s3);

    if (n1 + n2 != n3)
        return 0;

    int dp[n1 + 1][n2 + 1];

    dp[0][0] = 1;

    for (int i = 1; i <= n1; i++)
        dp[i][0] = dp[i - 1][0] &&
                   s1[i - 1] == s3[i - 1];

    for (int j = 1; j <= n2; j++)
        dp[0][j] = dp[0][j - 1] &&
                   s2[j - 1] == s3[j - 1];

    for (int i = 1; i <= n1; i++)
    {
        for (int j = 1; j <= n2; j++)
        {
            dp[i][j] = 0;

            if (s1[i - 1] == s3[i + j - 1])
                dp[i][j] |= dp[i - 1][j];

            if (s2[j - 1] == s3[i + j - 1])
                dp[i][j] |= dp[i][j - 1];
        }
    }

    return dp[n1][n2];
}

int main()
{
    char s1[] = "abc";
    char s2[] = "def";
    char s3[] = "adbcef";

    if (isInterleaving(s1, s2, s3))
        printf("True\n");
    else
        printf("False\n");

    return 0;
}
