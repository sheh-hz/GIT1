#include <stdio.h>
#include <string.h>

int isAnagram(char str1[], char str2[])
{
    int count[256] = {0};

    if (strlen(str1) != strlen(str2))
        return 0;

    for (int i = 0; str1[i] != '\0'; i++)
    {
        count[(unsigned char)str1[i]]++;
        count[(unsigned char)str2[i]]--;
    }

    for (int i = 0; i < 256; i++)
    {
        if (count[i] != 0)
            return 0;
    }

    return 1;
}

int main()
{
    char str1[] = "listen";
    char str2[] = "silent";

    if (isAnagram(str1, str2))
        printf("True\n");
    else
        printf("False\n");

    return 0;
}
