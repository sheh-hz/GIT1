#include <stdio.h>
#include <string.h>

int main()
{
    char str[] = "abcdefgh";
    int n = 4;
    int len = strlen(str);
    int part = len / n;

    for (int i = 0; i < len; i++)
    {
        printf("%c", str[i]);

        if ((i + 1) % part == 0)
            printf(" ");
    }

    return 0;
}
