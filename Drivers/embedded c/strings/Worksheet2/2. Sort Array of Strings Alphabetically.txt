#include <stdio.h>
#include <string.h>

int main()
{
    char a[3][20] = {"John", "Alice", "Bob"};
    char c[20];
    int i, j;

    for (i = 0; i < 3 - 1; i++)
    {
        for (j = i + 1; j < 3; j++)
        {
            if (strcmp(a[i], a[j]) > 0)
            {
                strcpy(c, a[i]);
                strcpy(a[i], a[j]);
                strcpy(a[j], c);
            }
        }
    }

    for (i = 0; i < 3; i++)
        printf("%s ", a[i]);

    return 0;
}
