#include <stdio.h>

int main()
{
    char a[] = "hello world";
    char b[] = "low";
    int c[256] = {0};
    int i, j = 0;

    for (i = 0; b[i] != '\0'; i++)
        c[(unsigned char)b[i]] = 1;

    for (i = 0; a[i] != '\0'; i++)
    {
        if (c[(unsigned char)a[i]] == 0)
            a[j++] = a[i];
    }

    a[j] = '\0';

    printf("%s\n", a);

    return 0;
}
