#include <stdio.h>

int main()
{
    char str[] = "abcabcbb";

    int last[256];

    for (int i = 0; i < 256; i++)
        last[i] = -1;

    int start = 0;
    int maxLen = 0;

    for (int i = 0; str[i] != '\0'; i++)
    {
        unsigned char ch = str[i];

        if (last[ch] >= start)
            start = last[ch] + 1;

        last[ch] = i;

        int len = i - start + 1;

        if (len > maxLen)
            maxLen = len;
    }

    printf("Length = %d\n", maxLen);

    return 0;
}
