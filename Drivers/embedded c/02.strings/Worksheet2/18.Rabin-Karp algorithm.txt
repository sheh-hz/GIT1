#include <stdio.h>
#include <string.h>

#define d 256
#define q 101

int main()
{
    char text[] = "abcxabcdabcdabcy";
    char pattern[] = "abcdabcy";

    int n = strlen(text);
    int m = strlen(pattern);

    int patternHash = 0;
    int textHash = 0;
    int h = 1;

    // h = pow(d, m-1) % q
    for (int i = 0; i < m - 1; i++)
        h = (h * d) % q;

    // Calculate initial hashes
    for (int i = 0; i < m; i++)
    {
        patternHash = (d * patternHash + pattern[i]) % q;
        textHash = (d * textHash + text[i]) % q;
    }

    for (int i = 0; i <= n - m; i++)
    {
        // Hash values match
        if (patternHash == textHash)
        {
            int j;

            for (j = 0; j < m; j++)
            {
                if (text[i + j] != pattern[j])
                    break;
            }

            if (j == m)
                printf("Pattern found at index %d\n", i);
        }

        // Calculate hash for next window
        if (i < n - m)
        {
            textHash = (d * (textHash - text[i] * h)
                        + text[i + m]) % q;

            if (textHash < 0)
                textHash += q;
        }
    }

    return 0;
}
