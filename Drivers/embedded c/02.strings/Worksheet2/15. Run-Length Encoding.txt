#include <stdio.h>

int main()
{
    char str[] = "aaabbcc";

    for (int i = 0; str[i] != '\0'; )
    {
        char ch = str[i];
        int count = 0;

        while (str[i] == ch)
        {
            count++;
            i++;
        }

        printf("%c%d", ch, count);
    }

    return 0;
}
