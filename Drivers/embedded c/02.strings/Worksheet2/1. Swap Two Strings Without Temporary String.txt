#include <stdio.h>
#include <string.h>

int main()
{
    char a[100] = "hello";
    char b[100] = "world";
    char c;
    int i, n;

    n = strlen(a);

    for (i = 0; i < n; i++)
    {
        c = a[i];
        a[i] = b[i];
        b[i] = c;
    }

    printf("str1 = %s\n", a);
    printf("str2 = %s\n", b);

    return 0;
}
