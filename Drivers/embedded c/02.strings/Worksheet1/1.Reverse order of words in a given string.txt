#include <stdio.h>
#include <string.h>
int main()
{
    char a[100];
    int i,s=0,e,j;
    scanf("%[^\n]s",a);
    int len=strlen(a);
    for(i=0;i<len/2;i++)
    {
        char temp=a[i];
        a[i]=a[len-i-1];
        a[len-i-1]=temp;
    }
    printf("%s\n",a);
    for (i = 0; i <= len; i++)
    {
        if (a[i] == ' ' || a[i] == '\0')
        {
            e = i - 1;

            for (j = s; j < (s + e + 1) / 2; j++)
            {
                char temp = a[j];
                a[j] = a[s + e - j];
                a[s + e - j] = temp;
            }

            s = i + 1;
        }
    }
    printf("%s",a);
    return 0;
}
