#include <stdio.h>
#include <string.h>
int main()
{
    char a[100],chara;
    int i,j,count=1,prev=0;
    scanf("%[^\n]s",a);
    
    int l=strlen(a);
    for (i = 0;a[i]!='\0'; i++)
    {
        count=1;
        for(j=i+1;j<l;j++)
        {
            if(a[i]==a[j])
            {
                count++;
            }
        }
        if(count>prev)
        {
            chara=a[i];
            prev=count;
        }
        
    }
    printf("%c appears %d times",chara,prev);
    return 0;
}
