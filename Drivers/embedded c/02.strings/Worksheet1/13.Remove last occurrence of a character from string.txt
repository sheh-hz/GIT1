#include <stdio.h>
#include <string.h>

int main()
{
    char a[200];
    int i, len,j;

    scanf("%[^\n]", a);
    len=strlen(a);
    for(i=len-1;i>0;i--)
    {
        if(a[i]=='s')
        {
            for(j=i;j<len;j++)
            {
                a[j]=a[j+1];
            }
            len--;
            break;
        }
        
    }

    printf("%s", a);

    return 0;
}
