#include <stdio.h>
#include <string.h>

int main()
{
    char a[100],i,count=0;
    scanf(" %[^\n]", a);
    

    for(i = 0; a[i] != '\0'; i++)
    {
        if((a[i] != ' ') && (i == 0 || a[i-1] == ' '))
        {
            count++;
        }
    }

    printf("Total words = %d", count);

    return 0;
}
