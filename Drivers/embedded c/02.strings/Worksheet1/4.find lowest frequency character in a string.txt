#include <stdio.h>
#include <string.h>

int main()
{
    char a[100];
    int i, j, count, min, l;

    scanf("%[^\n]", a);

    l = strlen(a);
    min = l;

    for (i = 0; i < l; i++)
    {
        for (j = 0; j < i; j++)
        {
            if (a[i] == a[j])
                break;
        }

        if (j != i)
            continue;

        count = 1;

        for (j = i + 1; j < l; j++)
        {
            if (a[i] == a[j])
                count++;
        }

        if (count < min)
            min = count;
    }

    for (i = 0; i < l; i++)
    {
        for (j = 0; j < i; j++)
        {
            if (a[i] == a[j])
                break;
        }

        if (j != i)
            continue;

        count = 1;

        for (j = i + 1; j < l; j++)
        {
            if (a[i] == a[j])
                count++;
        }

        if (count == min)
            printf("%c appears %d times\n", a[i], count);
    }

    return 0;
}
