#include <stdio.h>
#include <stdarg.h>

double average(int count, ...)
{
    va_list args;
    double sum = 0.0;

    va_start(args, count);

    for (int i = 0; i < count; i++)
    {
        double value = va_arg(args, double);

        sum += value;
    }

    va_end(args);

    return sum / count;
}

int main()
{
    double result;

    result = average(4, 10.5, 30.2, 40.1, 20.6);

    printf("The average is: %f\n", result);

    return 0;
}
