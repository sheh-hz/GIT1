#include <stdio.h>
#include <stdlib.h>
void fun(int *p)
{
    *p=20;
}
void main()
{
   int *p=NULL,i;
   p=malloc(sizeof(int)*5);
   if(p==NULL)
   {
       exit(1);
   }
   for(i=0;i<5;i++)
   {
       scanf("%d",p+i);
   }
   int *q=NULL;
   q=realloc(p,sizeof(int)*10);
   if(q==NULL)
   {
       exit(1);
   }
   for(i=5;i<10;i++)
   {
       scanf("%d",q+i);
   }
   for(i=0;i<10;i++)
   {
       printf("%d\t",*(q+i));
   }
   free(q);
}
