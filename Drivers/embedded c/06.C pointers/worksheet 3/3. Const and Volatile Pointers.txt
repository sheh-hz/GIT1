#include <stdio.h>
#include <stdint.h>

#define REG_BASE 0x40000000U

int main(void)
{
    volatile uint32_t *reg_modifiable =
        (volatile uint32_t *)REG_BASE;

    const uint32_t *reg_readonly =
        (const uint32_t *)REG_BASE;

    const volatile uint32_t *reg_const_volatile =
        (const volatile uint32_t *)REG_BASE;

    /*
     * Do NOT run this program on a normal PC.
     * These addresses are intended to represent MCU registers.
     */

    /* Read */
    uint32_t value1 = *reg_modifiable;
    uint32_t value2 = *reg_readonly;
    uint32_t value3 = *reg_const_volatile;

    printf("%u %u %u\n", value1, value2, value3);

    /* Write allowed */
    *reg_modifiable = 100;

    /* Write NOT allowed */
    // *reg_readonly = 200;
    // *reg_const_volatile = 300;

    return 0;
}
