#include <WiFi.h>
#include <WebServer.h>

// Wokwi Wi-Fi
const char* ssid = "Wokwi-GUEST";
const char* password = "";

// LED pin
#define LED_PIN 2

// Create web server on port 80
WebServer server(80);

// LED state
bool ledState = false;

// HTML webpage
String webpage()
{
  String html = R"rawliteral(
<!DOCTYPE html>
<html>

<head>
  <title>ESP32 LED Control</title>

  <meta name="viewport"
        content="width=device-width, initial-scale=1">

  <style>
    body {
      font-family: Arial;
      text-align: center;
      margin-top: 50px;
    }

    h1 {
      color: black;
    }

    button {
      padding: 15px 30px;
      font-size: 18px;
      margin: 10px;
      cursor: pointer;
    }
  </style>

</head>

<body>

  <h1>ESP32 LED Control</h1>

  <h2>LED Status: )rawliteral";

  if (ledState)
    html += "ON";
  else
    html += "OFF";

  html += R"rawliteral(</h2>

  <a href="/led/on">
    <button>TURN ON</button>
  </a>

  <a href="/led/off">
    <button>TURN OFF</button>
  </a>

</body>

</html>
)rawliteral";

  return html;
}

// Home page
void handleRoot()
{
  server.send(200, "text/html", webpage());
}

// Turn LED ON
void handleLEDOn()
{
  digitalWrite(LED_PIN, HIGH);

  ledState = true;

  server.send(200, "text/html", webpage());
}

// Turn LED OFF
void handleLEDOff()
{
  digitalWrite(LED_PIN, LOW);

  ledState = false;

  server.send(200, "text/html", webpage());
}

void setup()
{
  Serial.begin(115200);

  // Configure LED pin
  pinMode(LED_PIN, OUTPUT);

  digitalWrite(LED_PIN, LOW);

  // Connect to Wi-Fi
  WiFi.begin(ssid, password);

  Serial.print("Connecting to Wi-Fi");

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("Wi-Fi Connected!");

  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  // Define URLs
  server.on("/", handleRoot);

  server.on("/led/on", handleLEDOn);

  server.on("/led/off", handleLEDOff);

  // Start server
  server.begin();

  Serial.println("Web Server Started!");
}

void loop()
{
  server.handleClient();
}
